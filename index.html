              score += 30
              feedback_msg = "Benar! Bagus sekali."

            selected_card["y"] = selected_card["orig_y"]

                  c["valid"] for c in current_cards if c != selected_card

                  selected_card["x"] = selected_card["orig_x"]
                  selected_card["y"] = selected_card["orig_y"]
